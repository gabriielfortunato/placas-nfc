from flask import Flask, request, redirect, jsonify, render_template_string import sqlite3 app = Flask(_name_) # ========================================== # 1. FRONT-END (HTML + CSS + JS Integrados) # ========================================== HTML_PAINEL = """ Painel - Placas Inteligentes

Gestor Dinâmico de Placas NFC

Adicionar Nova Placa

Cliente Link do Chip (Rota) Destino Real Leituras Ações
""" # ========================================== # 2. BANCO DE DADOS E BACK-END # ========================================== def get_db(): conn = sqlite3.connect('placas.db') conn.row_factory = sqlite3.Row return conn def init_db(): conn = get_db() conn.execute('''CREATE TABLE IF NOT EXISTS placas (id INTEGER PRIMARY KEY AUTOINCREMENT, cliente TEXT NOT NULL, slug TEXT UNIQUE NOT NULL, destino TEXT NOT NULL, acessos INTEGER DEFAULT 0)''') conn.commit() conn.close() @app.route('/') def index(): # Renderiza o HTML diretamente da variável acima return render_template_string(HTML_PAINEL) @app.route('/api/placas', methods=['GET', 'POST']) def api_placas(): conn = get_db() if request.method == 'POST': data = request.json try: conn.execute('INSERT INTO placas (cliente, slug, destino) VALUES (?, ?, ?)', (data['cliente'], data['slug'], data['destino'])) conn.commit() return jsonify({'status': 'sucesso'}) except sqlite3.IntegrityError: return jsonify({'erro': 'Este slug/URL já está em uso.'}), 400 finally: conn.close() cur = conn.execute('SELECT * FROM placas ORDER BY id DESC') placas = [dict(row) for row in cur.fetchall()] conn.close() return jsonify(placas) @app.route('/api/placas/', methods=['DELETE']) def deletar_placa(slug): conn = get_db() conn.execute('DELETE FROM placas WHERE slug = ?', (slug,)) conn.commit() conn.close() return jsonify({'status': 'deletado'}) @app.route('/r/') def redirecionar(slug): conn = get_db() cur = conn.execute('SELECT destino FROM placas WHERE slug = ?', (slug,)) row = cur.fetchone() if row: conn.execute('UPDATE placas SET acessos = acessos + 1 WHERE slug = ?', (slug,)) conn.commit() conn.close() return redirect(row['destino']) conn.close() return "Placa não encontrada ou inativa.", 404 if _name_ == '_main_': init_db() app.run(debug=True, host='0.0.0.0', port=5000)
